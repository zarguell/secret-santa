   - Set textContent to "No wishlist set"
7. If wishlist text is non-empty:
   - Set textContent to the wishlist text
8. Handle errors with console.error() and set textContent to "Could not load wishlist"

Function signature:

```javascript
async function loadRecipientWishlist(recipientGuestId) {
  try {
    const response = await fetch(`/api/guest/${recipientGuestId}/wishlist`);
    const data = await response.json();
    const wishlistElement = document.getElementById("recipient-wishlist-text");
    if (data.wishlist === "") {
      wishlistElement.textContent = "No wishlist set";
    } else {
      wishlistElement.textContent = data.wishlist;
    }
  } catch (error) {
    console.error("Failed to load recipient wishlist:", error);
    document.getElementById("recipient-wishlist-text").textContent =
      "Could not load wishlist";
  }
}
```

Test structure:

```typescript
test("displays recipient wishlist after assignment revealed", async () => {
  // 1. Create party and guests
  const party = await createTestParty();
  const guest1 = await addGuestToParty(party.id, "santa@example.com", "Santa");
  const guest2 = await addGuestToParty(
    party.id,
    "recipient@example.com",
    "Recipient",
  );

  // 2. Assign gifts
  await assignGifts(party.id);

  // 3. Set wishlist for recipient
  const recipientWishlist = "I would love a book and some socks!";
  await fetch(`${BASE_URL}/api/guest/${guest2.id}/wishlist`, {
    method: "PUT",
    headers: { "Content-Type": "application/json" },
    body: JSON.stringify({ wishlist: recipientWishlist }),
  });

  // 4. Visit guest link and reveal assignment
  const guestLink = `${BASE_URL}/guest/${guest1.id}`;
  const page = await browser.newPage();
  await page.goto(guestLink);

  // 5. Reveal assignment
  await page.click("#reveal-btn");
  await page.waitForSelector("#recipient-wishlist-section", { visible: true });

  // 6. Verify recipient wishlist section is visible
  const section = await page.$("#recipient-wishlist-section");
  expect(section).not.toBeNull();

  // 7. Verify recipient wishlist text is displayed
  const wishlistText = await page.$eval(
    "#recipient-wishlist-text",
    (el) => el.textContent,
  );
  expect(wishlistText).toContain(recipientWishlist);

  // 8. Verify "No wishlist set" NOT shown
  expect(wishlistText).not.toContain("No wishlist set");

  await page.close();
});
```
